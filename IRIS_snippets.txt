# ===========================================
# convert time series
## convert to lower frequency (eg. from quarter to annual) 
CPI_A = convert(d.CPI_Q, 'y', Inf, 'method', @mean)

- 'mean' = mean of the values from all periods which fall into the desired period
- 'first' = value of the first period
- 'last' = value of the last period

## convert to higher frequency (eg. from annual to quarter)
CPI_Q = convert(d.CPI_A, 'q', Inf, 'method', 'quadavg')

- 'cubic' = cubic interpolation with low frequency value in the middle of the higher frequency grid
- 'quadsum' = quadratic interpolation such that the sum of the interpolated values matches the value of the original series in the respective lower frequency period
- 'quadavg' = quadratic interpolation such that average of the interpolated values matches value of the original series in the repsective lower frequency period
